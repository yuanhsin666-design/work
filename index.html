function generateSchedule() { const year = parseInt(document.getElementById('yearSelect').value); const month = parseInt(document.getElementById('monthSelect').value); const totalDays = getDaysInMonth(year, month); let scheduleData = Array.from({ length: staffs.length }, () => Array(totalDays).fill("休")); let shiftCounts = staffs.map(() => ({})); // 初始化统计数据（如果日期被锁定，保留锁定的数据） for (let day = 1; day <= totalDays; day++) { if (isDayLocked(year, month, day) && lastGeneratedSchedule && lastGeneratedSchedule[0][day - 1] !== undefined) { staffs.forEach((_, sIdx) => { let currentShift = lastGeneratedSchedule[sIdx][day - 1]; scheduleData[sIdx][day - 1] = currentShift; if (currentShift && currentShift !== "休" && currentShift !== "休(预)") { shiftCounts[sIdx][currentShift] = (shiftCounts[sIdx][currentShift] || 0) + 1; } }); } } // 从 1 号到月底逐日分配 for (let day = 1; day <= totalDays; day++) { if (isDayLocked(year, month, day) && lastGeneratedSchedule && lastGeneratedSchedule[0][day - 1] !== undefined) { continue; } // 获取当天未休假的人员列表 let availableStaffIndexes = []; staffs.forEach((staff, idx) => { if (!staff.customRests.includes(day)) { availableStaffIndexes.push(idx); } }); let workingCount = availableStaffIndexes.length; let todayShifts = []; if (workingCount <= baseShifts.length) { todayShifts = baseShifts.slice(0, workingCount); } else { todayShifts = [...baseShifts]; let extraSupportCount = workingCount - baseShifts.length; for (let s = 0; s < extraSupportCount; s++) { todayShifts.push(supportShift); } } let unassignedStaff = [...availableStaffIndexes]; todayShifts.forEach(shiftObj => { let shiftName = shiftObj.name; let isSupport = (shiftName === supportShift.name); let isCurrentHigh = (shiftObj.level === "高"); if (unassignedStaff.length === 0) return; unassignedStaff.sort((a, b) => { let countA = shiftCounts[a][shiftName] || 0; let countB = shiftCounts[b][shiftName] || 0; // 🔴 避开连续高强度（最高优先级硬性约束） if (isCurrentHigh && day > 1) { let yesterdayA = scheduleData[a][day - 2]; let yesterdayB = scheduleData[b][day - 2]; let aIsConsecutiveHigh = isHighShift(yesterdayA); let bIsConsecutiveHigh = isHighShift(yesterdayB); if (aIsConsecutiveHigh !== bIsConsecutiveHigh) { return aIsConsecutiveHigh ? 1 : -1; // 昨天排过高强度的排到最后 } } // 核心控制法则：岗位次数极小优先（严格避免单人单岗位超标） if (countA !== countB) { return countA - countB; } // 次要控制法则：月度总工作天数多的优先平衡 let totalWorkA = Object.values(shiftCounts[a]).reduce((sum, v) => sum + v, 0); let totalWorkB = Object.values(shiftCounts[b]).reduce((sum, v) => sum + v, 0); if (totalWorkA !== totalWorkB) return totalWorkA - totalWorkB; return Math.random() - 0.5; }); let chosenStaffIdx = unassignedStaff.shift(); scheduleData[chosenStaffIdx][day - 1] = shiftName; shiftCounts[chosenStaffIdx][shiftName] = (shiftCounts[chosenStaffIdx][shiftName] || 0) + 1; }); } lastGeneratedSchedule = scheduleData; renderAll(); saveDataToLocal(); document.getElementById('status').innerText = `${year}年${month}月 排班完毕（已解决连排与次数过高问题）！`; }
